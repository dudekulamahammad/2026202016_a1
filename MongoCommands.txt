# ------------------------------------------------------------
# 7. WORKFLOW 3 — $geoNear (nearest available vehicle within 5km)
#    Prints the 5 closest available vehicles.
#    NOTE: if the .js file does not print, use the eval fallback below.
# ------------------------------------------------------------
mongosh --quiet --file mongo/02_workflow3_geonear.js

# Fallback if the above prints nothing:
mongosh --quiet --eval '
const db = db.getSiblingDB("ridesync_db");
db.TelemetryPings.aggregate([
  { $geoNear: {
      near: { type: "Point", coordinates: [-73.9857, 40.7484] },
      distanceField: "dist_meters",
      maxDistance: 5000,
      spherical: true,
      query: { isAvailable: true }
  }},
  { $sort: { dist_meters: 1 } },
  { $limit: 5 },
  { $project: {
      _id: 0,
      vehicleId: 1,
      distance_km: { $divide: ["$dist_meters", 1000] },
      location: 1
  }}
]).forEach(printjson);
'

# ------------------------------------------------------------
# 8. WORKFLOW 4 — $facet (review analytics)
#    Rating distribution, top 10 feedback tags, average rating.
# ------------------------------------------------------------
mongosh --quiet --file mongo/03_workflow4_facet.js

# Fallback if the above prints nothing:
mongosh --quiet --eval '
const db = db.getSiblingDB("ridesync_db");
db.TripReviews.aggregate([
  { $facet: {
      rating_distribution: [
        { $group: { _id: "$rating", count: { $sum: 1 } } },
        { $sort: { _id: -1 } }
      ],
      common_feedback: [
        { $unwind: "$feedback_tags" },
        { $group: { _id: "$feedback_tags", total_occurrences: { $sum: 1 } } },
        { $sort: { total_occurrences: -1 } },
        { $limit: 10 }
      ],
      overall_average_rating: [
        { $group: { _id: null, average_rating: { $avg: "$rating" } } },
        { $project: { _id: 0, average_rating: 1 } }
      ]
  }}
]).forEach(printjson);
'

# ------------------------------------------------------------
# 9. EXPLAIN proof — Workflow 3 uses the 2dsphere index
# ------------------------------------------------------------
mongosh --quiet --eval '
const db = db.getSiblingDB("ridesync_db");
const explain = db.TelemetryPings.explain("executionStats").aggregate([
  { $geoNear: {
      near: { type: "Point", coordinates: [-73.9857, 40.7484] },
      distanceField: "dist_meters",
      maxDistance: 5000,
      spherical: true,
      query: { isAvailable: true }
  }},
  { $limit: 5 }
]);
const stats = explain.stages[0].$geoNearCursor.executionStats;
print("Index used   :", explain.stages[0].$geoNearCursor.queryPlanner.winningPlan.inputStage.indexName);
print("nReturned    :", stats.nReturned);
print("Execution ms :", stats.executionTimeMillis);
'

# ------------------------------------------------------------
# 10. EXPLAIN proof — Workflow 4 ($facet)
# ------------------------------------------------------------
mongosh --quiet --eval '
const db = db.getSiblingDB("ridesync_db");
const explain = db.TripReviews.explain("executionStats").aggregate([
  { $facet: {
      rating_distribution: [
        { $group: { _id: "$rating", count: { $sum: 1 } } },
        { $sort: { _id: -1 } }
      ],
      common_feedback: [
        { $unwind: "$feedback_tags" },
        { $group: { _id: "$feedback_tags", total_occurrences: { $sum: 1 } } },
        { $sort: { total_occurrences: -1 } },
        { $limit: 10 }
      ],
      overall_average_rating: [
        { $group: { _id: null, average_rating: { $avg: "$rating" } } },
        { $project: { _id: 0, average_rating: 1 } }
      ]
  }}
]);
const stats = explain.stages[0].$cursor.executionStats;
print("nReturned    :", stats.nReturned);
print("DocsExamined :", stats.totalDocsExamined);
print("Execution ms :", stats.executionTimeMillis);
'
